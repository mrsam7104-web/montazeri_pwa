-- ماشین‌بازی منتظری: ثبت جایزه ۵٪ و نمایش امن درخواست‌های پلاک‌های موجود در پنل مدیر
create table if not exists public.race_discount_claims (
  id bigint generated by default as identity primary key,
    full_name text not null,
      plate text not null,
        tracking_code text not null,
          is_registered boolean not null default false,
            discount_percent integer not null default 5 check (discount_percent = 5),
              created_at timestamptz not null default now()
              );
              alter table public.race_discount_claims enable row level security;
              drop policy if exists "admins read race discount claims" on public.race_discount_claims;
              create policy "admins read race discount claims" on public.race_discount_claims
               for select to authenticated using (true);
               revoke all on public.race_discount_claims from anon, authenticated;
               grant select on public.race_discount_claims to authenticated;
               grant insert, select on public.race_discount_claims to service_role;

               create or replace function public.race_norm_plate(v text)
               returns text language sql immutable as $$
                select upper(regexp_replace(translate(coalesce(v,''),'۰۱۲۳۴۵۶۷۸۹٠١٢٣٤٥٦٧٨٩','01234567890123456789'),'[^A-Za-z0-9آ-ی]','','g'));
                $$;

                create or replace function public.claim_race_discount(p_full_name text, p_plate text)
                returns jsonb
                language plpgsql
                security definer
                set search_path = public
                as $$
                declare
                  normalized text;
                    registered boolean := false;
                      code text;
                        claim_id bigint;
                        begin
                          if length(trim(coalesce(p_full_name,''))) < 2 or length(trim(coalesce(p_plate,''))) < 5 then
                              return jsonb_build_object('ok',false,'message','نام یا پلاک معتبر نیست.');
                                end if;
                                  normalized := public.race_norm_plate(p_plate);
                                    select exists (
                                        select 1 from public.workshop_state ws,
                                              jsonb_array_elements(coalesce(ws.cars,'[]'::jsonb)) car
                                                  where public.race_norm_plate(car->>'plate') = normalized
                                                    ) into registered;
                                                      code := lpad((floor(random()*10000))::int::text,4,'0');
                                                        if not registered then
                                                            -- Avoid issuing an easily-confused code that matches a current workshop tracking code.
                                                                while exists (
                                                                      select 1 from public.workshop_state ws,
                                                                              jsonb_array_elements(coalesce(ws.cars,'[]'::jsonb)) car
                                                                                    where car->>'trackingCode' = code
                                                                                        ) loop
                                                                                              code := lpad((floor(random()*10000))::int::text,4,'0');
                                                                                                  end loop;
                                                                                                    else
                                                                                                        code := '—';
                                                                                                          end if;
                                                                                                            insert into public.race_discount_claims(full_name,plate,tracking_code,is_registered,discount_percent)
                                                                                                              values (left(trim(p_full_name),100),left(trim(p_plate),30),code,registered,5)
                                                                                                                returning id into claim_id;
                                                                                                                  return jsonb_build_object('ok',true,'is_registered',registered,'tracking_code',code,'claim_id',claim_id);
                                                                                                                  exception when others then
                                                                                                                    return jsonb_build_object('ok',false,'message','ثبت انجام نشد؛ جدول workshop_state و ساختار cars را بررسی کنید.');
                                                                                                                    end;
                                                                                                                    $$;
                                                                                                                    revoke all on function public.claim_race_discount(text,text) from public;
                                                                                                                    grant execute on function public.claim_race_discount(text,text) to anon, authenticated;

-- این فایل از دو بخش تشکیل شده:
-- 1) SQL قبلی ارسال‌شده توسط کاربر، بدون تغییر در متن آن
-- 2) SQL گردونه تخفیف با وزن‌های متعادل‌شده
--
-- هشدار:
-- این فایل ترکیبی است و بخش SQL قبلی را هم دوباره اجرا می‌کند.
-- اگر SQL قبلی قبلاً روی پروژه اجرا شده، برای کاهش ریسک معمولاً فقط بخش گردونه را اجرا کنید.
-- قبل از اجرای کل فایل روی دیتابیس فعال، از داده‌ها و ساختار دیتابیس نسخه پشتیبان بگیرید.

      returns jsonb
      language plpgsql

      as $$
      declare
        row_car jsonb;
          wanted_plate text;

                begin
                  wanted_plate := replace(replace(replace(translate(lower(coalesce(p_plate,'')), '۰۱۲۳۴۵۶۷۸۹٠١٢٣٤٥٦٧٨٩', '01234567890123456789'), ' ', ''), '-', ''), 'ـ', '');

                        -- بررسی تک‌تک پرونده‌ها بدون افشای فهرست مشتریان
                          for row_car in select value from public.workshop_state s, jsonb_array_elements(s.cars) as value where s.id=1 loop
                              got_plate := replace(replace(replace(translate(lower(coalesce(row_car->>'plate','')), '۰۱۲۳۴۵۶۷۸۹٠١٢٣٤٥٦٧٨٩', '01234567890123456789'), ' ', ''), '-', ''), 'ـ', '');

                                      if got_plate = wanted_plate and got_phone = wanted_phone then return row_car; end if;
                                        end loop;
                                          return null;
                                          end;
                                          $$;

  -- صافکاری منتظری: جستجوی مشتری با پلاک + کد پیگیری چهاررقمی
-- در Supabase > SQL Editor اجرا کنید.
create or replace function public.lookup_vehicle_by_code(
  p_plate text,
  p_code text
)
returns jsonb
language plpgsql
security definer
set search_path = public
as $$
declare
  car jsonb;
  wanted_plate text;
  wanted_code text;
  stored_plate text;
  stored_code text;
begin
  wanted_plate := regexp_replace(
    translate(lower(coalesce(p_plate,'')),
      '۰۱۲۳۴۵۶۷۸۹٠١٢٣٤٥٦٧٨٩',
      '01234567890123456789'),
    '[[:space:]ـ-]', '', 'g');
  wanted_code := regexp_replace(
    translate(lower(coalesce(p_code,'')),
      '۰۱۲۳۴۵۶۷۸۹٠١٢٣٤٥٦٧٨٩',
      '01234567890123456789'),
    '[[:space:]ـ-]', '', 'g');

  if wanted_plate = '' or wanted_code !~ '^[0-9]{4}$' then
    return null;
  end if;

  for car in
    select value
    from public.workshop_state s,
         jsonb_array_elements(coalesce(s.cars, '[]'::jsonb)) as value
    where s.id = 1
  loop
    stored_plate := regexp_replace(
      translate(lower(coalesce(car->>'plate','')),
        '۰۱۲۳۴۵۶۷۸۹٠١٢٣٤٥٦٧٨٩',
        '01234567890123456789'),
      '[[:space:]ـ-]', '', 'g');
    stored_code := regexp_replace(
      translate(lower(coalesce(car->>'trackingCode','')),
        '۰۱۲۳۴۵۶۷۸۹٠١٢٣٤٥٦٧٨٩',
        '01234567890123456789'),
      '[[:space:]ـ-]', '', 'g');

      create or replace function public.lookup_vehicle(p_plate text, p_phone text)

      language plpgsql

        row_car jsonb;
          wanted_plate text;
            wanted_phone text;
              got_plate text;
                got_phone text;
                begin
                  wanted_plate := replace(replace(replace(translate(lower(coalesce(p_plate,'')), '۰۱۲۳۴۵۶۷۸۹٠١٢٣٤٥٦٧٨٩', '01234567890123456789'), ' ', ''), '-', ''), 'ـ', '');

                                      if got_plate = wanted_plate and got_phone = wanted_phone then return row_car; end if;

                                          end;
